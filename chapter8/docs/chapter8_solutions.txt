	int i, fact = 1;
	for (int i = 1; i <= n; i++) {
		fact = fact*i;
	}
	return fact;
}

	printf("Value of %d to the power of %d is %d\n", a, b, pow);
	return 0;

int power(int a, int b) {
	int i, res = 1;
	for (i = 0; i < b; i++) {
		res = res * a;
	}
	return res;
}
